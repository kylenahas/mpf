{{- /* Markdown links like [x](/build/modes/) are written from the site root. Prefix them with the site's base path so they still work when the site is served from a subfolder (GitHub Pages serves a project site at //). relURL keeps a leading slash relative to the host, so the slash is removed first. */ -}} {{- $dest := .Destination -}} {{- if and (hasPrefix $dest "/") (not (hasPrefix $dest "//")) -}} {{- $dest = strings.TrimPrefix "/" $dest | relURL -}} {{- end -}} {{- /* Links that leave the site open in a new tab, get class "ext" (an icon after them, in site.css) and a note for screen readers. */ -}} {{- if or (hasPrefix $dest "http://") (hasPrefix $dest "https://") -}} {{ .Text | safeHTML }} (external site, opens in a new tab) {{- else -}} {{ .Text | safeHTML }} {{- end -}} {{- /* no trailing newline */ -}}
